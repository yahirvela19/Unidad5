1"""Adaptador del paquete de pruebas del docente (datos/paquete_docente/*.csv). 

2 

3Traduce el modelo propuesto por el paquete (operación, recurso, RECIBIDA…ENTREGADA) al dominio del BRD 

4(Orden de Servicio, Trabajo, estados del glosario) y ejecuta sus casos con los servicios reales. 

5 

6Las reglas del paquete que NO están en el BRD (actor activo, cliente preferente, promoción) se evalúan aquí 

7con Specifications propias del paquete; no forman parte del dominio de AutoFix Pro. 

8Ver docs/testing/Diferencias_paquete_docente.md. 

9"""

90def leer_paquete(carpeta: Path) -> PaqueteDocente: 

91 actores = { 

92 a["actor_id"]: Actor(a["actor_id"], a["nombre"], a["es_preferente"] == "1", a["activo"] == "1") 

93 for a in _leer(carpeta, "actores.csv") 

94 } 

95 recursos = { 

96 r["recurso_id"]: Recurso( 

97 r["recurso_id"], 

98 r["descripcion"], 

99 Monto.de(r["precio_mxn"]), 

100 int(r["capacidad_o_stock"]), 

101 r["habilitado"] == "1", 

102 ) 

103 for r in _leer(carpeta, "catalogo.csv") 

104 } 

105 operaciones = { 

106 o["operacion_id"]: Operacion( 

107 o["operacion_id"], actores[o["actor_id"]], o["autorizado"] == "1", o["promocion_activa"] == "1" 

108 ) 

109 for o in _leer(carpeta, "operaciones.csv") 

110 } 

111 for linea in _leer(carpeta, "detalle_operaciones.csv"): 

112 operaciones[linea["operacion_id"]].lineas.append((recursos[linea["recurso_id"]], int(linea["cantidad"]))) 

113 return PaqueteDocente( 

114 actores, recursos, operaciones, _leer(carpeta, "casos_prueba.csv"), _leer(carpeta, "transiciones_prueba.csv") 

115 )

162 return ActorActivoSpecification().y(DetalleNoVacioSpecification()).y(AutorizacionRegistradaSpecification())

171def nuevo_contenedor(paquete: PaqueteDocente) -> Contenedor: 

172 reloj = RelojDelSistema() 

173 c = construir_contenedor(repositorios_en_memoria(), reloj, NotificacionSimulada(reloj), DespachadorEnMemoria()) 

174 for actor in paquete.actores.values(): 

175 c.repos.clientes.guardar(Cliente(actor.id, actor.nombre)) 

176 return c

222def completar(c: Contenedor, orden_id: str, operacion: Operacion) -> None: 

223 """Ejecuta los trabajos; un recurso no habilitado o sin existencia pasa a «En espera de refacción» (RN-07).""" 

224 orden = c.repos.ordenes.obtener_por_id(orden_id) 

225 recursos = {f"{r.descripcion} ({r.id})": (r, cantidad) for r, cantidad in operacion.lineas} 

226 faltantes = False 

227 for trabajo in orden.todos_los_trabajos(): 

228 recurso, cantidad = recursos[trabajo.descripcion] 

229 if not recurso.habilitado or cantidad > recurso.existencia: 

230 c.marcar_espera_de_refaccion.confirmar_falta_de_existencia( 

231 COMPRAS, orden_id, trabajo.id, recurso.id, "Proveedor del paquete" 

232 ) 

233 faltantes = True 

234 if faltantes: 

235 return # La orden queda «En espera de refacción» hasta registrar la recepción (RN-07, RN-08). 

236 for trabajo in c.repos.ordenes.obtener_por_id(orden_id).todos_los_trabajos(): 

237 if trabajo.estado == EstadoTrabajo.AUTORIZADO: 237 ↛ 236line 237 didn't jump to line 236 because the condition on line 237 was always true

238 c.ejecutar_trabajo.iniciar(TECNICO, orden_id, trabajo.id) 

239 c.ejecutar_trabajo.terminar(TECNICO, orden_id, trabajo.id) 

240 total = c.repos.ordenes.obtener_por_id(orden_id).total_cobrable() 

241 c.cerrar_orden.cerrar(CerrarOrdenDeServicioCommand(CAJA, orden_id, str(total.valor)))

252def ejecutar_transicion(paquete: PaqueteDocente, caso: dict) -> ResultadoTransicion: 

253 operacion = paquete.operaciones[caso["operacion_id"]] 

254 accion = caso["accion_sugerida"] 

255 c = nuevo_contenedor(paquete) 

256 orden = registrar(c, operacion) 

257 try: 

258 if accion == "saltar_estado": 

259 # Recibido → Cotizado sin pasar por Diagnosticado. 

260 c.generar_cotizacion.generar_y_enviar(ASESOR, orden.id, [{"descripcion": "Salto", "precio_unitario": "1"}]) 

261 elif accion == "cancelar": 

262 # El BRD no tiene cancelación desde «Recibido»; el rechazo total solo existe tras cotizar. 

263 transicionar_orden(c.repos.ordenes.obtener_por_id(orden.id).estado, EstadoOrden.CERRADA_SIN_REPARACION) 

264 else: 

265 cotizacion = cotizar(c, orden.id, operacion) 

266 if accion in ("aprobar", "repetir_aprobacion", "completar"): 266 ↛ 270line 266 didn't jump to line 270 because the condition on line 266 was always true

267 if not operacion.autorizado: 

268 raise ErrorDeDominio("La operación no tiene autorización del cliente; la orden sigue en espera.") 

269 autorizar(c, orden.id, cotizacion, operacion) 

270 if accion == "repetir_aprobacion": 

271 autorizar(c, orden.id, cotizacion, operacion) 

272 if accion == "completar": 

273 completar(c, orden.id, operacion) 

274 resultado = "Sin excepción" 

275 aceptada = True 

276 except ErrorDeDominio as error: 

277 resultado = f"{type(error).__name__}: {error.mensaje}" 

278 aceptada = False 

279 final = c.repos.ordenes.obtener_por_id(orden.id) 

280 eventos = [e.tipo for e in final.historial] 

281 if aceptada: 

282 resultado = f"Evento final: {eventos[-1]}" 

283 return ResultadoTransicion( 

284 final.estado.value, 

285 EQUIVALENCIA_DE_ESTADOS.get(final.estado, "(sin equivalente)"), 

286 resultado, 

287 aceptada, 

288 )

291def evaluar_caso(paquete: PaqueteDocente, caso: dict) -> tuple[str, str]: 

292 operacion = paquete.operaciones[caso["operacion_id"]] 

293 if caso["tipo_prueba"] == "validar_operacion": 

294 cumple = puede_procesarse().es_satisfecha_por(operacion) 

295 fallas = [ 

296 nombre 

297 for nombre, spec in ( 

298 ("actor inactivo", ActorActivoSpecification()), 

299 ("detalle vacío", DetalleNoVacioSpecification()), 

300 ("sin autorización", AutorizacionRegistradaSpecification()), 

301 ) 

302 if not spec.es_satisfecha_por(operacion) 

303 ] 

304 evidencia = f"total {operacion.total}" + (f"; no cumple: {', '.join(fallas)}" if fallas else "") 

305 return ("PASA" if cumple else "NO PASA"), evidencia 

306 cumple = beneficio_de_referencia().es_satisfecha_por(operacion) 

307 evidencia = ( 

308 f"preferente={'sí' if operacion.actor.preferente else 'no'}; total {operacion.total}; " 

309 f"promoción={'sí' if operacion.promocion_activa else 'no'}" 

310 ) 

311 return ("APLICA" if cumple else "NO APLICA"), evidencia

314def escribir_resultados(paquete: PaqueteDocente, carpeta_salida: Path) -> tuple[Path, Path]: 

315 carpeta_salida.mkdir(parents=True, exist_ok=True) 

316 casos = carpeta_salida / "resultados_casos_prueba.csv" 

317 with open(casos, "w", encoding="utf-8-sig", newline="") as archivo: 

318 escritor = csv.writer(archivo) 

319 escritor.writerow( 

320 ["caso_id", "operacion_id", "tipo_prueba", "regla_a_verificar", "resultado_alumno", "evidencia"] 

321 ) 

322 for caso in paquete.casos: 

323 resultado, evidencia = evaluar_caso(paquete, caso) 

324 escritor.writerow( 

325 [ 

326 caso["caso_id"], 

327 caso["operacion_id"], 

328 caso["tipo_prueba"], 

329 caso["regla_a_verificar"], 

330 resultado, 

331 evidencia, 

332 ] 

333 ) 

334 transiciones = carpeta_salida / "resultados_transiciones.csv" 

335 with open(transiciones, "w", encoding="utf-8-sig", newline="") as archivo: 

336 escritor = csv.writer(archivo) 

337 escritor.writerow( 

338 [ 

339 "caso_id", 

340 "operacion_id", 

341 "accion_sugerida", 

342 "estado_inicial", 

343 "resultado_alumno", 

344 "estado_final_brd", 

345 "estado_equivalente_paquete", 

346 "evento_o_excepcion", 

347 "precondicion_puede_procesarse", 

348 ] 

349 ) 

350 for caso in paquete.transiciones: 

351 r = ejecutar_transicion(paquete, caso) 

352 operacion = paquete.operaciones[caso["operacion_id"]] 

353 escritor.writerow( 

354 [ 

355 caso["caso_id"], 

356 caso["operacion_id"], 

357 caso["accion_sugerida"], 

358 caso["estado_inicial"], 

359 "ACEPTADA" if r.aceptada else "RECHAZADA", 

360 r.estado_final, 

361 r.equivalente_paquete, 

362 r.evento_o_excepcion, 

363 "PASA" if puede_procesarse().es_satisfecha_por(operacion) else "NO PASA", 

364 ] 

365 ) 

366 return casos, transiciones
